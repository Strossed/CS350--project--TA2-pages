package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.Set;
import java.util.TreeMap;
import java.util.TreeSet;
import org.apache.poi.ss.usermodel.CellStyle;
import org.apache.poi.ss.util.WorkbookUtil;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

/** Writes the course worksheets of a detailed enrollment report. */
public final class DetailedProjectionReport {
    private DetailedProjectionReport() {
    }

    /**
     * Imports one snapshot through the existing course loader and writes its worksheets.
     *
     * @param snapshot the input CSV already selected by the caller
     * @param output destination Excel workbook
     * @throws IOException if the snapshot cannot be read or the workbook cannot be saved
     * @throws IllegalArgumentException if the output would replace the input or courses are invalid
     */
    public static void writeSnapshot(Path snapshot, Path output) throws IOException {
        if (snapshot.toAbsolutePath().normalize().equals(output.toAbsolutePath().normalize())
                || (Files.exists(output) && Files.isSameFile(snapshot, output))) {
            throw new IllegalArgumentException("The workbook destination must differ from the snapshot");
        }
        write(output, SummaryProjection.loadCourses(snapshot));
    }

    /**
     * Writes exactly one alphabetically ordered worksheet for each course.
     * Sheets contain current enrollment and capacity supplied by the course model.
     * Projection curves and graphs belong to separate user stories.
     *
     * @param output destination .xlsx file; its parent directory must exist
     * @param courses distinct courses to include in the workbook
     * @throws IOException if the workbook cannot be saved
     * @throws IllegalArgumentException if the course list is empty or sheet names are invalid
     */
    public static void write(Path output, List<Course> courses) throws IOException {
        var sortedCourses = validatedCourses(courses);
        try (var workbook = new XSSFWorkbook()) {
            var heading = headingStyle(workbook);
            for (Course course : sortedCourses.values()) {
                addCourseSheet(workbook, course, heading);
            }
            try (var destination = Files.newOutputStream(output)) {
                workbook.write(destination);
            }
        }
    }

    private static TreeMap<String, Course> validatedCourses(List<Course> courses) {
        if (courses.isEmpty()) {
            throw new IllegalArgumentException("No courses are available for the detailed report");
        }
        var sortedCourses = new TreeMap<String, Course>();
        Set<String> sheetNames = new TreeSet<>(String.CASE_INSENSITIVE_ORDER);
        for (Course course : courses) {
            String name = course.getName();
            WorkbookUtil.validateSheetName(name);
            if (!sheetNames.add(name)) {
                throw new IllegalArgumentException("Duplicate course worksheet: " + name);
            }
            sortedCourses.put(name, course);
        }
        return sortedCourses;
    }

    private static CellStyle headingStyle(XSSFWorkbook workbook) {
        var font = workbook.createFont();
        font.setBold(true);
        var style = workbook.createCellStyle();
        style.setFont(font);
        return style;
    }

    private static void addCourseSheet(XSSFWorkbook workbook, Course course, CellStyle heading) {
        var sheet = workbook.createSheet(course.getName());
        var header = sheet.createRow(0);
        String[] columns = {"Course", "Current enrollment", "Capacity"};
        for (int column = 0; column < columns.length; column++) {
            var cell = header.createCell(column);
            cell.setCellValue(columns[column]);
            cell.setCellStyle(heading);
            sheet.setColumnWidth(column, 22 * 256);
        }
        var values = sheet.createRow(1);
        values.createCell(0).setCellValue(course.getName());
        values.createCell(1).setCellValue(course.getCurrentEnrollment());
        values.createCell(2).setCellValue(course.getCapacity());
        sheet.createFreezePane(0, 1);
    }
}
