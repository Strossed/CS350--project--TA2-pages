package edu.odu.cs.cs350;

import java.nio.file.Path;
import java.io.IOException;

/**
 * Entry point for the Enrollment Projection program.
 **/
public class Main {

    /**
     * Run the program.
     * @param args empty for the current summary, or --details SNAPSHOT.csv OUTPUT.xlsx
     * @throws IOException if the detailed report input or output cannot be accessed
     **/
    public static void main(String[] args) throws IOException {
        if (args.length != 0) {
            if (args.length != 3 || !args[0].equals("--details")) {
                throw new IllegalArgumentException("Usage: --details SNAPSHOT.csv OUTPUT.xlsx");
            }
            DetailedProjectionReport.writeSnapshot(Path.of(args[1]), Path.of(args[2]));
            System.out.println("Detailed report saved to " + args[2]);
            return;
        }
        //TEMPORARY: hard codes csv file path. Later will be replaced with command line argument parsing.
        Path csvFilePath = Path.of("src", "main", "resources", "summary", "202410", "2024-09-03.csv");
        SummaryProjection summaryProjection = new SummaryProjection(csvFilePath);

        System.out.println(summaryProjection.toString());
    }
}
