	/**
     * Imports and aggregates a dated course snapshot.
     *
     * The CSV must contain SUBJ, CRSE, XLST CAP, and
     * ENR columns. Each row is an offering; enrollment and capacity are
     * summed by subject and course number. A single snapshot has no projection
     * information, so projected enrollment is initialized to current enrollment.
     *
     * @param csvFile the dated course snapshot to import.
     * @return courses sorted by their subject and course number.
     * @throws IOException if the file cannot be read or required columns are absent.
     */
    public static List<Course> loadCourses(Path csvFile) throws IOException {
        Map<String, Integer> headers;
        Map<String, int[]> totals = new TreeMap<>();

        try (BufferedReader reader = Files.newBufferedReader(csvFile)) {
            String headerLine = reader.readLine();
            if (headerLine == null) {
                throw new IOException("Course snapshot is empty: " + csvFile);
            }
            headers = indexHeaders(parseCsvLine(headerLine));
            int subjIndex = requiredColumn(headers, "SUBJ");
            int crseIndex = requiredColumn(headers, "CRSE");
            int capacityIndex = requiredColumn(headers, "XLST CAP");
            int enrollmentIndex = requiredColumn(headers, "ENR");
            int lineNumber = 1;
            String line;
            while ((line = reader.readLine()) != null) {
                lineNumber++;
                if (line.isBlank()) {
                    continue;
                }

                String[] fields = parseCsvLine(line);
                int lastRequiredIndex = Math.max(
                    Math.max(subjIndex, crseIndex),
                    Math.max(capacityIndex, enrollmentIndex));
                if (fields.length <= lastRequiredIndex) {
                    continue;
                }

                String subject = fields[subjIndex].trim();
                String courseNumber = fields[crseIndex].trim();
                if (subject.isEmpty() || courseNumber.isEmpty()) {
                    continue;
                }

                int capacity = parseInteger(fields[capacityIndex], csvFile, lineNumber);
                int enrollment = parseInteger(fields[enrollmentIndex], csvFile, lineNumber);
                String name = subject + courseNumber;
                int[] courseTotals = totals.computeIfAbsent(name, ignored -> new int[2]);
                courseTotals[0] = Math.addExact(courseTotals[0], enrollment);
                courseTotals[1] = Math.addExact(courseTotals[1], capacity);
            }
        }

        List<Course> courses = new ArrayList<>();
        for (Map.Entry<String, int[]> entry : totals.entrySet()) {
            int enrollment = entry.getValue()[0];
            courses.add(new Course(entry.getKey(), enrollment, enrollment, entry.getValue()[1]));
        }
        return courses;
    }

    /**
     * Extracts a sorted list of unique course identifiers from a CSV file.
     *
     * The CSV must contain a header row with columns named SUBJ (subject)
     * and CRSE (course number). Each subsequent row represents a course
     * offering. The method concatenates SUBJ and CRSE (e.g., CS115)
     * for each row, deduplicates the results, and returns them in ascending order.
     *
     * @param csvFile the path to the CSV file containing course data.
     * @return a sorted list of unique course names.
     * @throws IOException if an I/O error occurs while reading the file.
     */
    public static List<String> extractUniqueCourseNames(Path csvFile) throws IOException {
        Set<String> uniqueNames = new TreeSet<>();
        try (BufferedReader br = Files.newBufferedReader(csvFile)) {
            String headerLine = br.readLine();
            if (headerLine == null) {
                return new ArrayList<>(uniqueNames);
            }
            Map<String, Integer> headers = indexHeaders(parseCsvLine(headerLine));
            Integer subjIdx = headers.get("SUBJ");
            Integer crseIdx = headers.get("CRSE");
            if (subjIdx == null || crseIdx == null) {
                return new ArrayList<>(uniqueNames);
            }
            String line;
            while ((line = br.readLine()) != null) {
                if (line.isBlank()) continue;
                String[] parts = parseCsvLine(line);
                if (parts.length <= Math.max(subjIdx, crseIdx)) continue;
                String subj = parts[subjIdx].trim();
                String crse = parts[crseIdx].trim();
                if (!subj.isEmpty() && !crse.isEmpty()) {
                    uniqueNames.add(subj + crse);
                }
            }
        }
        return new ArrayList<>(uniqueNames);
    }

    private static Map<String, Integer> indexHeaders(String[] headers) {
        Map<String, Integer> indexes = new HashMap<>();
        for (int i = 0; i < headers.length; i++) {
            String header = headers[i].trim();
            if (i == 0) {
                header = header.replaceFirst("^\uFEFF", "");
            }
            indexes.put(header.toUpperCase(Locale.ROOT), i);
        }
        return indexes;
    }

    private static int requiredColumn(Map<String, Integer> headers, String name) throws IOException {
        Integer index = headers.get(name);
        if (index == null) {
            throw new IOException("Course snapshot is missing required column: " + name);
        }
        return index;
    }

    private static int parseInteger(String value, Path csvFile, int lineNumber) throws IOException {
        try {
            return Integer.parseInt(value.trim());
        } catch (NumberFormatException exception) {
            throw new IOException("Invalid integer in " + csvFile + " at line " + lineNumber, exception);
        }
    }

    private static String[] parseCsvLine(String line) throws IOException {
        List<String> fields = new ArrayList<>();
        StringBuilder field = new StringBuilder();
        boolean inQuotes = false;
        for (int i = 0; i < line.length(); i++) {
            char character = line.charAt(i);
            if (character == '"') {
                if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    field.append('"');
                    i++;
                } else {
                    inQuotes = !inQuotes;
                }
            } else if (character == ',' && !inQuotes) {
                fields.add(field.toString());
                field.setLength(0);
            } else {
                field.append(character);
            }
        }
        if (inQuotes) {
            throw new IOException("Unterminated quoted field in CSV line");
        }
        fields.add(field.toString());
        return fields.toArray(new String[0]);
    }
