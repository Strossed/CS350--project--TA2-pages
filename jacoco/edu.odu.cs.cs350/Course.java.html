package edu.odu.cs.cs350;

/**
 * Represents a course in the system.
 */
public class Course {
    private final String name;
    private final int currentEnrollment;
    private final int projectedEnrollment;
    private final int capacity;

    /**
     * Creates a course with its enrollment and capacity values.
     */
    public Course(String name, int currentEnrollment, int projectedEnrollment, int capacity) {
        this.name = name;
        this.currentEnrollment = currentEnrollment;
        this.projectedEnrollment = projectedEnrollment;
        this.capacity = capacity;
    }

    public String getName() {
        return name;
    }

    public int getCurrentEnrollment() {
        return currentEnrollment;
    }

    public int getProjectedEnrollment() {
        return projectedEnrollment;
    }

    public int getCapacity() {
        return capacity;
    }
      /**
     * Determine whether the projected enrollment exceeds the total cap.
     * @return true if projected enrollment is greater than capacity
     **/
    public boolean exceedsCap() {
        return projectedEnrollment > capacity;
    }
}
